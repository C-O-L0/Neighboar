SELECT b.building_id, b.block, b.street,
       n.hawker_id, n.name, n.address, n.status, n.stall_count, n.latitude, n.longitude, n.dist, c.within_1km
FROM building b
JOIN town tw ON tw.town_id = b.town_id
CROSS JOIN LATERAL (
    SELECT h.hawker_id, h.name, h.address, h.status, h.stall_count, h.latitude, h.longitude, 2 * 6378100 * asin(sqrt(
    sin((radians(h.latitude::double precision) - radians(b.latitude::double precision)) / 2) ^ 2
    + cos(radians(b.latitude::double precision)) * cos(radians(h.latitude::double precision))
      * sin((radians(h.longitude::double precision) - radians(b.longitude::double precision)) / 2) ^ 2)) AS dist
    FROM hawker_centre h
    WHERE h.status <> %(status)s
    ORDER BY dist, h.hawker_id
    LIMIT 1
) n
CROSS JOIN LATERAL (
    SELECT count(*)::int AS within_1km
    FROM hawker_centre h
    WHERE h.status <> %(status)s AND 2 * 6378100 * asin(sqrt(
    sin((radians(h.latitude::double precision) - radians(b.latitude::double precision)) / 2) ^ 2
    + cos(radians(b.latitude::double precision)) * cos(radians(h.latitude::double precision))
      * sin((radians(h.longitude::double precision) - radians(b.longitude::double precision)) / 2) ^ 2)) <= 1000
) c
WHERE tw.name = %(town)s AND b.latitude IS NOT NULL
ORDER BY n.dist, b.building_id
