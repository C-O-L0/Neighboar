WITH o AS (SELECT latitude, longitude FROM building WHERE building_id = %(place)s AND latitude IS NOT NULL),
per_block AS (
    SELECT t.building_id,
           count(*)::int AS n,
           percentile_disc(0.5) WITHIN GROUP (ORDER BY t.resale_price) AS median_price,
           percentile_disc(0.5) WITHIN GROUP (ORDER BY t.storey_min + t.storey_max) AS storey_sum,
           percentile_disc(0.5) WITHIN GROUP (ORDER BY t.floor_area_sqm) AS area_median
    FROM resale_transaction t
    
    WHERE t.flat_type_id = %(flat_type_id)s AND t.txn_month >= %(start)s AND t.txn_month <= %(end)s
    GROUP BY t.building_id
),
fits AS (
    SELECT p.building_id, b.block, b.street, tw.name AS town, p.n, p.median_price, p.storey_sum,
           p.area_median, b.lease_commence_year AS lease_year, 2 * 6378100 * asin(sqrt(
    sin((radians(b.latitude::double precision) - radians(o.latitude::double precision)) / 2) ^ 2
    + cos(radians(o.latitude::double precision)) * cos(radians(b.latitude::double precision))
      * sin((radians(b.longitude::double precision) - radians(o.longitude::double precision)) / 2) ^ 2)) AS distance, b.latitude, b.longitude
    FROM per_block p
    JOIN building b ON b.building_id = p.building_id
    JOIN town tw ON tw.town_id = b.town_id
    CROSS JOIN o
    WHERE p.median_price <= %(budget)s AND b.latitude IS NOT NULL AND 2 * 6378100 * asin(sqrt(
    sin((radians(b.latitude::double precision) - radians(o.latitude::double precision)) / 2) ^ 2
    + cos(radians(o.latitude::double precision)) * cos(radians(b.latitude::double precision))
      * sin((radians(b.longitude::double precision) - radians(o.longitude::double precision)) / 2) ^ 2)) <= %(radius)s
)
SELECT c.total, r.building_id, r.block, r.street, r.town, r.n, r.median_price, r.storey_sum, r.area_median,
       r.lease_year, r.distance, r.latitude, r.longitude
FROM (SELECT count(*)::int AS total FROM fits) c
LEFT JOIN LATERAL (
    SELECT * FROM fits ORDER BY distance, building_id LIMIT %(limit)s OFFSET %(offset)s
) r ON true
ORDER BY r.distance, r.building_id
